cmake_minimum_required(VERSION 3.20)
project(bisect VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(BUILD_SHARED_LIBS ON)
find_package(ECM 6.26 REQUIRED NO_MODULE)
set(CMAKE_MODULE_PATH ${ECM_MODULE_PATH} ${CMAKE_MODULE_PATH})
include(KDEInstallDirs)
find_package(KWin REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Qml)
find_package(KF6CoreAddons REQUIRED)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endif()
configure_file(src/metadata.json ${CMAKE_BINARY_DIR}/metadata.json COPYONLY)
kcoreaddons_add_plugin(bisect
    INSTALL_NAMESPACE kwin/effects
    SOURCES src/effect.cpp src/qtmetadata.qrc
)
target_include_directories(bisect PRIVATE src ${CMAKE_BINARY_DIR})
target_link_libraries(bisect PRIVATE KWin::kwin Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Qml)
set_target_properties(bisect PROPERTIES PREFIX "")
